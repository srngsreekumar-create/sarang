forecast_table


get_metrics = function(actual, predicted) {
  error = actual - predicted
  ME = mean(error)
  RMSE = sqrt(mean(error^2))
  MAE = mean(abs(error))
  MAPE = mean(abs(error / actual)) * 100
  
  return(c(
    ME = ME,
    RMSE = RMSE,
    MAE = MAE,
    MAPE = MAPE
  ))
}
get_metrics

results = t(sapply(predictions, function(predicted) {
  get_metrics(actual, predicted)
}))

results = data.frame(
  Model = rownames(results),
  results,
  row.names = NULL
)
results

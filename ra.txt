#validating and tesing slr

faithful

scatter.smooth(x=faithful$eruptions,y=faithful$waiting,
               main="waiting time v/s eruption duration",
               xlab="eruption duration",ylab = "waiting time")

cor(faithful$eruptions,faithful$waiting)

l=lm(waiting~eruptions,data = faithful)
l

model_summary=summary(l)
model_summary

model_coeffs=model_summary$coefficients
model_coeffs

nrow(faithful)

beta.estimate=model_coeffs["eruptions","Estimate"]
beta.estimate

std.error=model_coeffs["eruptions","Std. Error"]
std.error

t_value=beta.estimate/std.error
t_value

alpha = 0.05  # Change to 0.5 if you strictly need a 50% confidence level
n = 272       # Replace 272 with your actual sample size if different

t_table = qt(alpha/2, df = n-2, lower.tail = FALSE)
t_table

f=summary(l)$fstatistic
f

f_table=qf(0.95,df1=f[2],df=f[3])
f_table

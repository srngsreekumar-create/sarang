data = data.frame(
  HWC = c(1,1.5,2,3,4,4.5,5,5.5,6,6.5,7,8,9,10,11,12,13,14,15),
  TS = c(6.3,11.1,20,24,26.1,30,33.8,34,38.1,39.9,42,46.1,53.1,52,52.5,48,42.8,27.8,21.9)
)

write.csv(data, "TensileStrength.csv", row.names=FALSE)

plot(data$HWC, data$TS, xlab="HWC", ylab="TS")

model1 = lm(TS ~ HWC + I(HWC^2), data=data)
summary(model1)
model2 = lm(TS ~ poly(HWC, 2, raw=TRUE), data=data)
summary(model2)

HWC_val = seq(0, 20, 0.1)
TS_val = predict(model2, data.frame(HWC=HWC_val))

plot(data$HWC, data$TS, xlab="HWC", ylab="TS")
lines(HWC_val, TS_val, col="green", lwd=2)

start_vals = list(a=max(data$TS), b=data$HWC[which.max(data$TS)])

model3 = nls(TS ~ a*exp(-(HWC-b)^2/(2*c^2)),
             data=data, start=c(start_vals, c=2))
summary(model3)

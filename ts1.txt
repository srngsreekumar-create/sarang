model

summary(model)

# Overall F-test
summary(model)$fstatistic

vif(model)

v <- vif(model)


if (any(v > 10)) {
  print("Serious multicollinearity: VIF > 10")
} else if (any(v > 5)) {
  print("Warning: Multicollinearity present (VIF > 5)")
} else {
  print("No serious multicollinearity: All VIF values <= 5")
}
